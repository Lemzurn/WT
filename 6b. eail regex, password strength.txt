<html>
<body>
    Email: <input id="email">
    Pass: <input id="pass" onkeyup="check()">
    <span id="msg"></span>
    <script>
        function check() {
            var e = document.getElementById("email").value;
            var p = document.getElementById("pass").value;
            var atPos = e.indexOf("@");
            var dotPos = e.lastIndexOf(".");
            
            if (atPos > 0 && dotPos > atPos + 1 && dotPos < e.length - 1 && !e.includes(" ")) {
                if (p.length > 8) document.getElementById("msg").innerText = "Strong";
                else document.getElementById("msg").innerText = "Weak";
            } else {
                document.getElementById("msg").innerText = "Enter valid email first";
            }
        }
    </script>
</body>
</html>
